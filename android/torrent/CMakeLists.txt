cmake_minimum_required(VERSION 3.20)
project(storm_torrent LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT STORM_LIBTORRENT_SOURCE OR NOT BOOST_ROOT)
  message(FATAL_ERROR "Provide pinned STORM_LIBTORRENT_SOURCE and BOOST_ROOT; see build_torrent.py")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(build_tests OFF CACHE BOOL "" FORCE)
set(build_examples OFF CACHE BOOL "" FORCE)
set(build_tools OFF CACHE BOOL "" FORCE)
set(python-bindings OFF CACHE BOOL "" FORCE)
set(logging OFF CACHE BOOL "" FORCE)
set(i2p OFF CACHE BOOL "" FORCE)
set(Boost_NO_BOOST_CMAKE ON)
set(OPENSSL_USE_STATIC_LIBS ON)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenSSL OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_GnuTLS ON)
set(CMAKE_DISABLE_FIND_PACKAGE_LibGcrypt ON)
find_package(OpenSSL 3.5 REQUIRED)
add_subdirectory("${STORM_LIBTORRENT_SOURCE}" libtorrent EXCLUDE_FROM_ALL)
if(ANDROID)
  # aligned_alloc was added in API 28; Boost.Align uses posix_memalign on API 24.
  target_compile_definitions(torrent-rasterbar PUBLIC BOOST_ASIO_DISABLE_STD_ALIGNED_ALLOC)
endif()
add_library(storm_torrent SHARED torrent_session.cpp)
target_link_libraries(storm_torrent PRIVATE torrent-rasterbar)
target_compile_options(storm_torrent PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
  target_sources(storm_torrent PRIVATE jni.cpp)
  set_target_properties(storm_torrent PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  target_link_options(storm_torrent PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
else()
  enable_testing()
  add_executable(torrent_tests ../../tests/torrent/test_session.cpp)
  target_link_libraries(torrent_tests PRIVATE storm_torrent torrent-rasterbar)
  add_test(NAME torrent_metadata_and_local_transfer COMMAND torrent_tests)
  # Compile the production JNI for both a real JVM checker and VM fault injection.
  find_package(Java 17 REQUIRED COMPONENTS Runtime Development)
  # Only headers are needed: a headless JDK need not provide FindJNI's AWT library.
  get_filename_component(STORM_JAVAC_REAL "${Java_JAVAC_EXECUTABLE}" REALPATH)
  get_filename_component(STORM_JAVA_BIN "${STORM_JAVAC_REAL}" DIRECTORY)
  get_filename_component(STORM_JAVA_ROOT "${STORM_JAVA_BIN}" DIRECTORY)
  find_path(STORM_JNI_INCLUDE jni.h PATHS "${STORM_JAVA_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_path(STORM_JNI_PLATFORM_INCLUDE jni_md.h PATHS "${STORM_JAVA_ROOT}/include/linux" NO_DEFAULT_PATH REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  target_sources(storm_torrent PRIVATE jni.cpp)
  target_include_directories(storm_torrent PRIVATE ${STORM_JNI_INCLUDE} ${STORM_JNI_PLATFORM_INCLUDE})
  add_executable(torrent_jni_faults ../../tests/torrent/test_jni_faults.cpp)
  target_include_directories(torrent_jni_faults PRIVATE ${STORM_JNI_INCLUDE} ${STORM_JNI_PLATFORM_INCLUDE})
  add_test(NAME torrent_jni_fault_injection COMMAND torrent_jni_faults)
  add_test(NAME torrent_jvm_check COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/torrent/run_jvm.py
    --library $<TARGET_FILE:storm_torrent>
    --java ${Java_JAVA_EXECUTABLE} --javac ${Java_JAVAC_EXECUTABLE})
endif()
