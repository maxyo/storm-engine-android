cmake_minimum_required(VERSION 3.20)
project(storm_archive LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(S "${STORM_ARCHIVE_SOURCES}")
set(P "${STORM_ARCHIVE_PREFIX}")
if(NOT EXISTS "${P}/lib/libarchive.a" OR NOT EXISTS "${S}/innoextract-1.9/src/setup/info.cpp")
 message(FATAL_ERROR "Use android/archive/build.py to prepare pinned readers")
endif()
set(B "${S}/boost_1_90_0")
file(GLOB FS_SOURCES "${B}/libs/filesystem/src/*.cpp")
list(FILTER FS_SOURCES EXCLUDE REGEX "windows_")
add_library(storm_boost STATIC ${FS_SOURCES}
 "${B}/libs/iostreams/src/zlib.cpp" "${B}/libs/iostreams/src/gzip.cpp"
 "${B}/libs/iostreams/src/bzip2.cpp")
target_include_directories(storm_boost PUBLIC "${B}" "${P}/include")
target_compile_definitions(storm_boost PRIVATE BOOST_FILESYSTEM_STATIC_LINK BOOST_FILESYSTEM_SOURCE
 BOOST_FILESYSTEM_HAS_STAT_ST_MTIM BOOST_FILESYSTEM_HAS_POSIX_AT_APIS BOOST_FILESYSTEM_HAS_CXX20_ATOMIC_REF)
set(I "${S}/innoextract-1.9")
foreach(var ALIGNOF STD_UNIQUE_PTR BUILTIN_BSWAP16 BUILTIN_BSWAP32 BUILTIN_BSWAP64 LZMA BUILTIN_CONV TIMEGM GMTIME_R)
 set(INNOEXTRACT_HAVE_${var} 1)
endforeach()
configure_file("${I}/src/configure.hpp.in" "${CMAKE_CURRENT_BINARY_DIR}/configure.hpp")
file(GLOB INNO_SOURCES "${I}/src/crypto/*.cpp" "${I}/src/loader/*.cpp" "${I}/src/setup/*.cpp" "${I}/src/stream/*.cpp")
list(FILTER INNO_SOURCES EXCLUDE REGEX "arc4.cpp$")
add_library(storm_inno STATIC ${INNO_SOURCES}
 "${I}/src/util/console.cpp" "${I}/src/util/encoding.cpp" "${I}/src/util/load.cpp"
 "${I}/src/util/log.cpp" "${I}/src/util/time.cpp")
target_include_directories(storm_inno PUBLIC "${I}/src" "${CMAKE_CURRENT_BINARY_DIR}" "${B}" "${P}/include")
target_link_libraries(storm_inno PRIVATE storm_boost)
add_library(storm_archive SHARED archive.cpp manifest.cpp memory.cpp libarchive_backend.cpp inno_backend.cpp)
target_include_directories(storm_archive PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(storm_archive PRIVATE storm_inno storm_boost
 "${P}/lib/libarchive.a" "${P}/lib/liblzma.a" "${P}/lib/libbz2.a" "${P}/lib/libz.a")
target_link_options(storm_archive PRIVATE "-Wl,--no-undefined" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "-Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free,--wrap=strdup" "-Wl,-z,max-page-size=16384")
option(STORM_ARCHIVE_TESTING "Export private allocator fault controls for host tests" OFF)
if(STORM_ARCHIVE_TESTING)
 if(ANDROID)
  message(FATAL_ERROR "Fault injection must never be enabled in an Android artifact")
 endif()
 target_compile_definitions(storm_archive PRIVATE STORM_ARCHIVE_TESTING)
endif()
option(STORM_ARCHIVE_HOST_JNI "Build JNI bridge for host JVM tests" OFF)
if(STORM_ARCHIVE_HOST_JNI AND NOT ANDROID)
 find_package(JNI REQUIRED COMPONENTS JVM)
 target_include_directories(storm_archive PRIVATE ${JNI_INCLUDE_DIRS})
endif()
if(ANDROID OR STORM_ARCHIVE_HOST_JNI)
 target_sources(storm_archive PRIVATE jni.cpp)
endif()
if(NOT ANDROID)
 add_executable(storm_archive_cli tests/cli.cpp)
 target_link_libraries(storm_archive_cli PRIVATE storm_archive)
endif()
