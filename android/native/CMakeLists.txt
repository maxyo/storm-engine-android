cmake_minimum_required(VERSION 3.18)
project(storm_android_probe LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "This integration probe requires Android arm64-v8a")
endif()
set(STORM_ANDROID_DEPS "" CACHE PATH "Installed SDL dependency prefix")
if(NOT EXISTS "${STORM_ANDROID_DEPS}/lib/libSDL2.so")
    message(FATAL_ERROR "Build the real SDL dependency first and set STORM_ANDROID_DEPS")
endif()
add_library(SDL2 SHARED IMPORTED)
set_target_properties(SDL2 PROPERTIES IMPORTED_LOCATION "${STORM_ANDROID_DEPS}/lib/libSDL2.so"
    INTERFACE_INCLUDE_DIRECTORIES "${STORM_ANDROID_DEPS}/include")
add_library(storm_probe SHARED probe.cpp)
target_compile_features(storm_probe PRIVATE cxx_std_20)
target_include_directories(storm_probe PRIVATE ../../src/libs/util/include)
target_link_libraries(storm_probe PRIVATE SDL2 vulkan log)
target_link_options(storm_probe PRIVATE -Wl,--no-undefined)
