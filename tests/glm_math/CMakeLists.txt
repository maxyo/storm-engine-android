cmake_minimum_required(VERSION 3.18)
project(storm-glm-regression LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(root "${CMAKE_CURRENT_LIST_DIR}/../..")
include("${root}/cmake/StormSetup.cmake")
set(SRC_DIRS src)
set(PUBLIC_INCLUDE_DIRS include)
set(TESTSUITE_DIRS testsuite)
set(RESOURCE_DIRS rsrc)
set(SOURCE_FILE_EXTENSIONS "*.cpp")
set(HEADER_FILE_EXTENSIONS "*.h" "*.hpp")
add_subdirectory("${root}/src/libs/math" math)
add_executable(check check.cpp)
target_link_libraries(check PRIVATE math)
target_include_directories(check PRIVATE sdk_shape)
if(STORM_MATH_BASELINE_HEADERS)
    target_include_directories(check BEFORE PRIVATE "${STORM_MATH_BASELINE_HEADERS}")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(check PRIVATE -Wall -Wextra -ffp-contract=off)
    if(STORM_MATH_SANITIZE)
        target_compile_options(check PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(check PRIVATE -fsanitize=address,undefined)
    endif()
endif()
add_test(NAME glm_math_contract COMMAND check)

# Separate TU is essential: no GLM/standard header may be pre-included here.
add_executable(packed_vector_include packed_vector_include.cpp)
target_link_libraries(packed_vector_include PRIVATE math)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(packed_vector_include PRIVATE -Wall -Wextra -ffp-contract=off)
    if(STORM_MATH_SANITIZE)
        target_compile_options(packed_vector_include PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(packed_vector_include PRIVATE -fsanitize=address,undefined)
    endif()
endif()
add_test(NAME packed_vector_include COMMAND packed_vector_include)

add_executable(packed_check packed_check.cpp)
target_link_libraries(packed_check PRIVATE math)
target_include_directories(packed_check PRIVATE sdk_shape)
if(STORM_MATH_BASELINE_HEADERS)
    target_include_directories(packed_check BEFORE PRIVATE "${STORM_MATH_BASELINE_HEADERS}")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(packed_check PRIVATE -Wall -Wextra -ffp-contract=off)
    if(STORM_MATH_SANITIZE)
        target_compile_options(packed_check PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(packed_check PRIVATE -fsanitize=address,undefined)
    endif()
endif()
add_test(NAME packed_math_contract COMMAND packed_check)
